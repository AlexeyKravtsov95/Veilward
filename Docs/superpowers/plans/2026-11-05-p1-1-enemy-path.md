- [x] **Step 3: Implement minimal precomputed geometry**

`PathPosition` — readonly struct с validated finite X/Y и public getters. В `PathDefinition` сохранить `float[] _segmentLengths` и `float[] _cumulativeLengths`; constructor вычисляет Euclidean length каждого сегмента через `Math.Sqrt`. `GetPosition` использует binary search или линейный поиск по precomputed boundaries, clamp к `[0, TotalLength]`; вычисляет `t = offset / segmentLength` и линейную интерполяцию. Никаких новых collections в `GetPosition`.

```csharp
public PathPosition GetPosition(float distance)
{
    if (float.IsNaN(distance) || float.IsInfinity(distance))
        throw new ArgumentOutOfRangeException(nameof(distance));

    var clamped = Math.Max(0f, Math.Min(distance, TotalLength));
    var segment = 0;
    while (segment < _segmentLengths.Length - 1 && clamped > _cumulativeLengths[segment + 1])
        segment++;

    var start = Points[segment];
    var end = Points[segment + 1];
    var offset = clamped - _cumulativeLengths[segment];
    var t = offset / _segmentLengths[segment];
    return new PathPosition(start.X + (end.X - start.X) * t, start.Y + (end.Y - start.Y) * t);
}
```

- [x] **Step 4: Implement validated wave and allocation-free stepping**

`WaveDefinition` сохраняет validated enemy/count/firstSpawnTick/interval. `BattleState` сохраняет `List<EnemyState>` и один `ReadOnlyCollection` wrapper; добавить internal `SetPhase`, `AdvanceTick`, `AddEnemy`, `RemoveEnemyAt`, `ApplyLeak`. `EnemyState` получает internal `Advance(float distance)` с clamp по path в симуляции. Public API остаётся read-only.

`BattleSimulation.StartWave()` только из `Preparation`: `Combat`, затем `SpawnDueEnemies`. `Step()` только в `Combat`: advance tick, spawn due, forward movement loop с `RemoveAt(index)` без reverse event order. Distance = `Math.Min(TotalLength, old + Speed / 30f)`. Leak удаляет до damage. Обработка продолжается по ID даже при достижении 0; после loop `Defeat` приоритетнее `Victory`. Victory только если все spawns выполнены и active count 0.

```csharp
public void Step()
{
    if (State.Phase != BattlePhase.Combat) return;
    State.AdvanceTick();
    SpawnDueEnemies();

    var index = 0;
    while (index < State.Enemies.Count)
    {
        var enemy = State.Enemies[index];
        var distance = Math.Min(_path.TotalLength,
            enemy.DistanceAlongPath + enemy.Definition.Speed / SimulationTiming.TicksPerSecond);
        if (distance >= _path.TotalLength)
        {
            State.RemoveEnemyAt(index);
            State.ApplyLeak(enemy.Definition.LeakDamage);
            continue;
        }

        enemy.Advance(distance);
        index++;
    }

    if (State.BaseHealth == 0) State.SetPhase(BattlePhase.Defeat);
    else if (_nextSpawnIndex == _wave.Count && State.Enemies.Count == 0)
        State.SetPhase(BattlePhase.Victory);
}
```

```csharp
[MenuItem("Veilward/Build P1.1 Battle Scene")]
public static void Build()
{
    const string configPath = "Assets/_Game/Configs/TestWave.asset";
    const string prefabPath = "Assets/_Game/Prefabs/Shade.prefab";
    const string materialPath = "Assets/_Game/Art/Materials/GrayboxPath.mat";
    AssetDatabase.DeleteAsset(configPath);
    AssetDatabase.DeleteAsset(prefabPath);
    AssetDatabase.DeleteAsset(materialPath);

    var config = BattleScenarioConfig.CreateForTests();
    AssetDatabase.CreateAsset(config, configPath);
    var pathMaterial = new Material(Shader.Find("Universal Render Pipeline/Lit"));
    pathMaterial.color = new Color(0.25f, 0.55f, 0.5f);
    AssetDatabase.CreateAsset(pathMaterial, materialPath);

    var enemy = GameObject.CreatePrimitive(PrimitiveType.Capsule);
    enemy.name = "Shade";
    enemy.transform.localScale = new Vector3(0.6f, 0.6f, 0.6f);
    var enemyPrefab = PrefabUtility.SaveAsPrefabAsset(enemy, prefabPath);
    UnityEngine.Object.DestroyImmediate(enemy);

    var scene = EditorSceneManager.OpenScene("Assets/_Game/Scenes/Battle.unity");
    foreach (var root in scene.GetRootGameObjects()) UnityEngine.Object.DestroyImmediate(root);

    var ground = GameObject.CreatePrimitive(PrimitiveType.Cube);
    ground.name = "Ground";
    ground.transform.localScale = new Vector3(16f, 0.2f, 10f);
    ground.transform.position = new Vector3(0f, -0.15f, 0f);

    var points = new[] { new Vector2Int(-5, -2), new Vector2Int(0, -2), new Vector2Int(0, 2), new Vector2Int(5, 2) };
    for (var i = 0; i < points.Length - 1; i++)
    {
        var start = new Vector3(points[i].x, 0.05f, points[i].y);
        var end = new Vector3(points[i + 1].x, 0.05f, points[i + 1].y);
        var delta = end - start;
        var segment = GameObject.CreatePrimitive(PrimitiveType.Cube);
        segment.name = $"Path Segment {i}";
        segment.transform.position = (start + end) * 0.5f;
        segment.transform.rotation = Quaternion.LookRotation(delta.normalized);
        segment.transform.localScale = new Vector3(1.2f, 0.1f, delta.magnitude);
        segment.GetComponent<Renderer>().sharedMaterial = pathMaterial;
    }

    foreach (var marker in new[] { new Vector2Int(-5, -2), new Vector2Int(5, 2) })
    {
        var cylinder = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
        cylinder.name = marker == points[0] ? "Spawn Marker" : "Seal Heart";
        cylinder.transform.position = new Vector3(marker.x, 0.25f, marker.y);
        cylinder.transform.localScale = new Vector3(0.8f, 0.25f, 0.8f);
    }

    var cameraObject = new GameObject("Battle Camera", typeof(Camera));
    cameraObject.transform.position = new Vector3(8f, 10f, 8f);
    cameraObject.transform.LookAt(Vector3.zero);
    cameraObject.GetComponent<Camera>().orthographic = true;
    cameraObject.GetComponent<Camera>().orthographicSize = 8f;
    new GameObject("Directional Light", typeof(Light)).GetComponent<Light>().type = LightType.Directional;

    var canvas = new GameObject("HUD", typeof(Canvas), typeof(CanvasScaler), typeof(GraphicRaycaster));
    canvas.GetComponent<Canvas>().renderMode = RenderMode.ScreenSpaceOverlay;
    var textObject = new GameObject("Stats", typeof(Text));
    textObject.transform.SetParent(canvas.transform, false);
    var text = textObject.GetComponent<Text>();
    text.font = Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");
    text.fontSize = 24;
    text.rectTransform.anchorMin = new Vector2(0f, 1f);
    text.rectTransform.anchorMax = new Vector2(0f, 1f);
    text.rectTransform.anchoredPosition = new Vector2(140f, -40f);

    var controllerObject = new GameObject("Battle Controller", typeof(BattleSceneController));
    var serialized = new SerializedObject(controllerObject.GetComponent<BattleSceneController>());
    serialized.FindProperty("scenario").objectReferenceValue = config;
    serialized.FindProperty("enemyPrefab").objectReferenceValue = enemyPrefab.transform;
    serialized.FindProperty("hudText").objectReferenceValue = text;
    serialized.ApplyModifiedPropertiesWithoutUndo();

    EditorSceneManager.MarkSceneDirty(scene);
    EditorSceneManager.SaveScene(scene);
    AssetDatabase.SaveAssets();
}
```
