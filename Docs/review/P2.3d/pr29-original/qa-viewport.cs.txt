using System;
using System.Reflection;
using UnityEditor;
using UnityEngine;
var asm=typeof(Editor).Assembly;
var flags=BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Instance|BindingFlags.Static;
var gameType=asm.GetType("UnityEditor.GameView");
var view=EditorWindow.GetWindow(gameType);
var sizesType=asm.GetType("UnityEditor.GameViewSizes");
var singleton=typeof(ScriptableSingleton<>).MakeGenericType(sizesType).GetProperty("instance", flags).GetValue(null);
var groupType=gameType.GetProperty("currentSizeGroupType", flags).GetValue(null);
var group=sizesType.GetMethod("GetGroup", flags).Invoke(singleton,new[]{groupType});
var sizeType=asm.GetType("UnityEditor.GameViewSize");
var mode=Enum.Parse(asm.GetType("UnityEditor.GameViewSizeType"),"FixedResolution");
var size=Activator.CreateInstance(sizeType,flags,null,new object[]{mode,1600,900,"P23d 1600x900"},null);
group.GetType().GetMethod("AddCustomSize",flags).Invoke(group,new[]{size});
var index=(int)group.GetType().GetMethod("GetTotalCount",flags).Invoke(group,null)-1;
gameType.GetProperty("selectedSizeIndex",flags).SetValue(view,index);
view.Focus(); view.Repaint();
return new {index, group=groupType.ToString()};
